cmake_minimum_required(VERSION 3.20)
project(LiShenNative LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(LiShen WIN32 src/main.cpp app.rc)
target_compile_definitions(LiShen PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
  target_compile_options(LiShen PRIVATE /W4 /utf-8 /EHsc /MT)
  target_link_options(LiShen PRIVATE /MANIFEST:NO)
else()
  target_link_options(LiShen PRIVATE -municode)
endif()
target_link_libraries(LiShen PRIVATE user32 gdi32 gdiplus shell32)

enable_testing()
add_executable(animation_test tests/animation_test.cpp)
add_test(NAME animation COMMAND animation_test)
